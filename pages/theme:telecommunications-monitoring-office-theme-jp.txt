このテーマは電気通信監視室の記事向けに設計されています。Black Highlighterテーマを使用しているため、上記のとおりそちらもincludeしてください。

[[code type="css"]]
@supports(display: grid) {
    :root {
        /* ===S-CSS-P統合=== */
        /* 新しいCSSテーマを作成する場合は、少なくとも以下の3つの変数を含めてください。 */
        --theme-base: "black-highlighter";
        /* "black-highlighter"または"sigma9"のいずれかにしてください */
        --theme-id: "telecommunications-monitoring-office-theme";
        /* テーマページのURLに設定します。例: "component:ar-theme"なら"ar-theme"にします */
        --theme-name: "電気通信監視室テーマ";
        /* テーマの正式名称に設定します */

        /* ===ヘッダー要素=== */
        --logo-image: url("https://scp-jp-storage.wdfiles.com/local--files/file%3A5384396-443-cuyh/TMOwhite.png");
        --header-subtitle: "電気通信監視室";

        /* ===標準テーマカラー=== */
        --bright-accent: 183, 18, 52;
        --medium-accent: 183, 18, 52;
        --dark-accent: 183, 18, 52;

        /* ===ヘッダーグラデーション=== */
        --header-gradient-color-bottom: 183, 18, 52;

        /* ===主要グラデーション=== */
        --diagonal-stripes: unset;
    }
h3,h6 {
    color: darkred;
}
.blockquote {
    background: rgb(var(--white-monochrome)) !important;
    border: rgb(183, 18, 52) solid !important;
}
}
