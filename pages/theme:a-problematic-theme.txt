+ What's This?

This is A Problematic Theme by [[*user winkwonkboi]], designed for articles in [[[the-appliance-war-hub|The Appliance War]]] --canon-- series or within the general setting of Site-37.

On any wiki, just apply the following code at the top of the page:

[[div class="code"]]
@@[[include :scp-wiki:theme:black-highlighter-theme]]@@
@@[[include :scp-wiki:theme:a-problematic-theme version=1]]@@
[[/div]]

//Note: As this is a Black Highlighter based theme, the Black Highlighter theme should be used in addition for this theme to work. To learn more, go [[[theme:black-highlighter-theme|here.]]]//

----

+ Examples

A horizontal rule can be created using five hyphens @@"-----"@@ and stretches across the entire page unless it is placed inside anything (such as a blockquote). Lines separating sections of a page are horizontal rules.

++ Second Title
+++ Third Title
++++ Fourth Title
+++++ Fifth Title
++++++ Sixth Title

/* Standard Version */

@import url('https://fonts.bunny.net/css2?family=IBM+Plex+Sans+Condensed:wght@700&display=swap');
@import url('https://fonts.bunny.net/css2?family=Podkova:wght@400..800&display=swap');
@import url('https://fonts.bunny.net/css2?family=Open+Sans:ital,wght@0,300..800;1,300..800&display=swap');
@import url('https://fonts.bunny.net/css2?family=Nanum+Gothic+Coding&display=swap');

    :root {
        /* S-CSS-P INTEGRATION */
        --theme-base: "black-highlighter";
        --theme-id: "a-problematic-theme";
        --theme-name: "A Problematic Theme";

        /* Black & White */
        --white: 252, 252, 252;
        --darker: 48, 48, 52;
        --yet-darker: 27, 28, 31;
        --darkest: 20, 22, 24;
        --black: 0, 0, 0;

        /* STANDARD Colors */
        --main-green: 151, 206, 142;
        --sage-green: 103, 146, 103;
        --christmas-green: 0, 135, 62;
        --celadon: 172, 225, 175;
        --chetwode: 240, 255, 240;
        --artichoke: 75, 111, 68;

        /* DARK Colors */
        --gloomy-green: 74, 93, 35;
        --laurel-green: 169, 186, 157;
        --army-green: 75, 83, 32;
        --kombu-green: 53, 66, 48;
        --gin: 216, 228, 188;

        /* RED Colors */
        --bloody-stain: 186, 9, 42;
        --light-red: 255, 204, 203;
        --salmon: 250, 128, 114;
        --desire: 234, 60, 83;
        --maroon: 128, 0, 0;
        --sangria: 94, 25, 20;
        --prismatic: 208, 61, 51;
        --light-coral: 240, 128, 128;

        /* Fonts */
        --header-font: "IBM Plex Sans Condensed", sans-serif;
        --title-font: "Podkova", serif;
        --body-font: "Open Sans", sans-serif;
        --mono-font: "Nanum Gothic Coding", monospace;
        --UI-font: var(--title-font);

        /* Header Elements */
        --logo-image: url("https://scp-wiki.wdfiles.com/local--files/theme%3Aa-problematic-theme/standard37.svg");
        --header-title: "FOUNDATION SITE-37";
        --header-subtitle: "We've Got Problems";

        /* Header Gradient */
        --header-gradient-color-bottom: var(--main-green);
        --header-gradient-color-middle: var(--main-green);
        --header-gradient-color-top: var(--main-green);
        --diagonal-stripes: unset;

        /* Theme Colors */
        --bright-accent: var(--celadon);
        --medium-accent: var(--main-green);
        --dark-accent: var(--main-green);
        --gray-monochrome: var(--sage-green);
        --dark-gray-monochrome: var(--artichoke);

        --link-color: var(--christmas-green);
        --hover-link-color: var(--sage-green);
        --visited-link-color: var(--artichoke);

        /* Menu Colors */
        --swatch-menubg-color: var(--artichoke);
    }

    #header {
        --swatch-headerh1-color: var(--black);
        --swatch-headerh2-color: var(--black);
        --search-icon-color: var(--black);

        --login-line-divider-color: var(--black);
        --login-username-color: var(--black);
        --login-myaccount-color: var(--black);

        --login-dropdown-text-color: var(--black);
        --login-mobile-icon-color: var(--black);
    }
 
    #interwiki,
    #side-bar {
        --sidebar-media-bg-color: var(--chetwode);
        --sidebar-links-text: var(--white);
    }

    #main-content {
        --toc-body-link-color: var(--white);
        --toc-body-link-hover-color: var(--white);
        --toc-body-hover-arrow-color: var(--white);
    }
